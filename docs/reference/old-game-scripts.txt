-- Script Variables
local cd = 10
local db = false
local cost = 1
local beingheld = false
local lastTime
local currentTime
-- Remotes
local remotes = game.ReplicatedStorage.Remotes
local raceSkillEvent = remotes.CombatRemotes.Skills.RaceSkill
-- Anis
local breathinAni = Instance.new("Animation")
breathinAni.AnimationId = "rbxassetid://18292059208"
local BreathInTrack = animator:LoadAnimation(breathinAni)
local breathholdAni = Instance.new("Animation")
breathholdAni.AnimationId = "rbxassetid://18292087980"
local BreathHoldTrack = animator:LoadAnimation(breathholdAni)
local screamAni = Instance.new("Animation")
screamAni.AnimationId = "rbxassetid://18292761496"
local ScreamTrack = animator:LoadAnimation(screamAni)
-- Connections
local con1 = nil
local con2 = nil
skill.Activated:Connect(function()
	if db == false and MagoiBar.Value >= 1 then
		db = true
		beingheld = true
		raceSkillEvent:FireServer("BattleCry", "Stage 1")
		BreathInTrack:Play()
		BreathInTrack:AdjustSpeed(0.5)
		BreathInTrack.Stopped:Connect(function()
			BreathHoldTrack:Play()
		end)
		currentTime = os.clock()
	end
end)
skill.Deactivated:Connect(function()
	if beingheld == true then
		beingheld = false
		lastTime = os.clock()
		local timeIncrease = lastTime - currentTime
		if lastTime - currentTime <= 0.5 then
			raceSkillEvent:FireServer("BattleCry", "Stage 2", "Quick")
			BreathInTrack:Stop()
			BreathHoldTrack:Stop()
			ScreamTrack:Play()
			ScreamTrack:AdjustSpeed(0.3)
		elseif lastTime - currentTime > 0.5 then
			raceSkillEvent:FireServer("BattleCry", "Stage 2", nil, timeIncrease)
			BreathHoldTrack:Stop()
			ScreamTrack:Play()
			ScreamTrack:AdjustSpeed(0.3)
		end
		delay(cd, function()
			db = false
		end)
	end
end)
-- Forcing Deactivation if out of Magoi
MagoiBar.Changed:Connect(function()
	if MagoiBar.Value <= 0 then
		if beingheld == true then
			beingheld = false
			lastTime = os.clock()
			local timeIncrease = lastTime - currentTime
			if lastTime - currentTime <= 0.5 then
				raceSkillEvent:FireServer("BattleCry", "Stage 2", "Quick")
			elseif lastTime - currentTime > 0.5 then
				raceSkillEvent:FireServer("BattleCry", "Stage 2", nil, timeIncrease)
				BreathHoldTrack:Stop()
				ScreamTrack:Play()
				ScreamTrack:AdjustSpeed(0.3)
			end
			delay(cd, function()
				db = false
			end)
		end
	end
end)
-- Interruption via Hit
player.Character:FindFirstChild("Effects").ChildAdded:Connect(function(child)
	if child.Name == "LightHit" or child.Name == "MagicHit" then
		if beingheld == true then
			beingheld = false
			BreathHoldTrack:Stop()
			
			raceSkillEvent:FireServer("BattleCry", "Interrupt")
			
			delay(cd*1.5, function()
				db = false
			end)
		end
	end
end)
raceSkillEvent.OnClientEvent:Connect(function(check, target)
	if check == "ShowEnemy" then
		if not target:FindFirstChild("Locked On") then 
			local HL = Instance.new("Highlight")
			HL.Adornee = target
			HL.Name = "Locked On"
			HL.FillTransparency = 1
			HL.OutlineColor = Color3.fromRGB(255, 255, 255)
			HL.FillColor = Color3.fromRGB(0, 0, 0)
			HL.Parent = target
			delay(60, function()
				HL:Destroy()
			end)
		end
	end
end)

				until Using.Value == false

local function midecode(weaponcode)
	-- Calculate the total length of enchantments
	local rarity = string.sub(weaponcode, 10, 10)  -- Rarity is at a fixed position
	local numEnchantments = enchantmentSpots[rarity] or 0
	local enchantmentLength = numEnchantments * 2
	-- Remove the WeaponHealth and Enchantments parts from the code
	local mainPartLength = #weaponcode - (2 + enchantmentLength)
	local mainPart = string.sub(weaponcode, 1, mainPartLength)
	-- Find the position of the unique code (always 4 digits after the name)
	local uniqueCodePattern = "%d%d%d%d"
	local uniqueCodeStart, uniqueCodeEnd = string.find(mainPart, uniqueCodePattern)
	-- Extract the Name (before the unique code)
	local Name = string.sub(mainPart, 1, uniqueCodeStart - 1)
	-- Extract the unique code (4 digits)
	local UniqueCode = string.sub(mainPart, uniqueCodeStart, uniqueCodeEnd)
	-- Extract the Rarity (1 character after the unique code)
	local Rarity = string.sub(mainPart, uniqueCodeEnd + 1, uniqueCodeEnd + 1)
	-- Extract the rest of the string after the rarity
	local rest = string.sub(mainPart, uniqueCodeEnd + 2)
	-- Extract the weapon type
	local WeaponType
	for _, type in ipairs(weapontypes) do
		if string.find(rest, type) then
			WeaponType = type
			break
		end
	end

	local axis = axis or Vector3.new(0,1,0) --backup axis
	local dot, uxv = u:Dot(v), u:Cross(v)
	if (dot < -0.99999) then return CFrame.fromAxisAngle(axis, math.pi) end
	return CFrame.new(0, 0, 0, uxv.x, uxv.y, uxv.z, 1 + dot)

		SpellUpdate(player.OnCharacter.Spells.Value)
		Spells.Changed:Connect(function(s)
			SpellUpdate(s)
		end)
		-- Listening for Spell Words
		-- Setting Up List
		local WordsofPower = {}
		for _,v in Knowledge:GetChildren() do
			local W = string.lower(v.Name)
			table.insert(WordsofPower, W)
		end
		Knowledge.ChildAdded:Connect(function(v)
			if not table.find(WordsofPower, v.Name) then
				table.insert(WordsofPower, string.lower(v.Name))
			end
		end)
		-- Scanning Message
		player.Chatted:Connect(function(words, dm)
			local wordcopy = string.lower(words)
			local searchSpots = {}
			for _,v in powerfulwords do
				insert(searchSpots, string.find(wordcopy, v))
			end
			local numberofwords = #searchSpots/2
			local PowerWord1 = nil
			local PowerWord2 = nil
			local PowerWord3 = nil
			local ignore = false
			if #searchSpots > 2 then
				if searchSpots[3] - searchSpots[2] ~= 2 then
					ignore = true
				end
			end
			if #searchSpots > 4 then
				if searchSpots[5] - searchSpots[4] ~=2 then
					ignore = true
				end
			end
			for i=1,#searchSpots,2 do
				local sav = string.sub(wordcopy, searchSpots[i], searchSpots[i+1])
				if not PowerWord1 then
					PowerWord1 = sav
				elseif not PowerWord2 and not ignore then
					PowerWord2 = sav
				elseif not PowerWord3 and not ignore then
					PowerWord3 = sav
				end
			end
			local finalProduct
			if PowerWord1 then
				finalProduct = PowerWord1
				if PowerWord2 then
					finalProduct = finalProduct.." "..PowerWord2
					if PowerWord3 then
						finalProduct = finalProduct.." "..PowerWord3
					end
				end
			end
			if table.find(WordsofPower, finalProduct) then
				heatEvent:FireClient(player, finalProduct)
				waterEvent:FireClient(player, finalProduct)
				lightEvent:FireClient(player, finalProduct)
				soundEvent:FireClient(player, finalProduct)
				strengthEvent:FireClient(player, finalProduct)
				windEvent:FireClient(player,finalProduct)
			end
		end)
	end)

strengthEvent.OnServerEvent:Connect(function(player, word, stage, extra)
	-- Needed
	local scale = player.Stats.MaxMagoi.Value / 1000
	local function performRaycast(offset)
		local rayOrigin = extra:ToWorldSpace(CFrame.new(0, 5000, -15 - scale) * offset).Position
		local rayDirection = Vector3.new(0, -10000, 0)
		local param = RaycastParams.new()
		param.FilterDescendantsInstances = {workspace.Map}
		param.FilterType = Enum.RaycastFilterType.Include
		return workspace:Raycast(rayOrigin, rayDirection, param)
	end
	local function getNeededFrame(rayResult)
		local x, y, z = extra:ToOrientation()
		return CFrame.new(rayResult.Position) * CFrame.Angles(x, y, z)
	end
	local function performSideRaycast(centerPosition, direction)
		local rayDirection = direction * 1000
		local param = RaycastParams.new()
		param.FilterDescendantsInstances = {workspace.Map}
		param.FilterType = Enum.RaycastFilterType.Include
		return workspace:Raycast(centerPosition, rayDirection, param)
	end
	-- Important Variables
	local chara = player.Character
	-- Setting Up Wand
	local wand = chara:FindFirstChild("DisplayWand"):WaitForChild("Handle")
	if word == "Yalmas" then
		if stage == "Activation" then
			-- The initial activation of the spell, still waiting for player to either Composite or go through with the normal spell.
			--  Casting Val + Subtracting Magoi
			local Casting = Instance.new("StringValue")
			Casting.Name = "Casting"
			Casting.Value = word
			Casting.Parent = chara:FindFirstChild("Effects")
			chara:FindFirstChild("IntFold"):FindFirstChild("MagoiBar").Value  -= extra
			-- Particle and Sound Work
			wand.PartAtt.gravAura.Enabled = true
			wand.PartAtt.gravAura:Emit(3)
			wand.PartAtt.gYalmasravFissure.Enabled = true
			wand.PartAtt.gravFissure:Emit(3)
			wand.TrailAtt1.gravTrail.Enabled = true
			local c = chara.Torso.SoundEffects:FindFirstChild("MagicActivation1"):Clone()
			c.Parent = chara.Torso
			c:Play()
			c.Ended:Connect(function()
				c:Destroy()
			end)
			-- Slowing
			Speed.SpeedHandle(chara, 0.5, "Casting")
		elseif stage == "StrengthParticle" then
			-- Spawning Gravity Pool
			-- Set up positions for each rock and dust block
			local positions = {
				Center = {offset = CFrame.new(0, 0, -3), angle = 90},
				Right = {offset = CFrame.new(-16 - scale, 0, 2), angle = 115},
				Left = {offset = CFrame.new(16 + scale, 0, 2), angle = 65}
			}
			
			-- Prechecking
			local testResult = performRaycast(positions.Center.offset)
			if not testResult then
				-- If Not Ray, Ladder Creation
				print("Ladder needed.")
			else 
				for name, position in positions do
					local rockRayResult = performRaycast(position.offset)
					local neededFrame = getNeededFrame(rockRayResult)
					local sideRay
					local sideFrame
					-- Checking Distance
					local TooFar = false
					if name ~= "Center" then
						for specRock, specPos in positions do
							if specRock == "Center" then
								-- Location Center Rock
								local secondRay = performRaycast(specPos.offset)
								local secondFrame = getNeededFrame(secondRay)
								local mag = (neededFrame.Position - secondFrame.Position).Magnitude
								if mag > 30 then
									TooFar = true
									local dir
									if name == "Left" then
										dir = extra.RightVector
									else
										dir = -extra.RightVector
									end
									sideRay = performSideRaycast(secondFrame.Position, dir)
									
									if sideRay then
										sideFrame = getNeededFrame(sideRay)	
									end
								end
							end
						end
					end
					-- Create pool and lines
					local gravLines, gravPool = strengthEffects.GravityLines:Clone(), strengthEffects.StrengthPool:Clone()
					gravLines.Name = "GravLines"..name
					gravPool.Name = "GravPool"..name
					if TooFar and sideRay then 
						if name == "Left" then
							gravLines.CFrame = sideFrame:ToWorldSpace(CFrame.new(0, 0, 0) * CFrame.Angles(math.rad(90), 0, math.rad(90)))
							gravPool.CFrame = sideFrame:ToWorldSpace(CFrame.new(-0.5, 5, 0) * CFrame.Angles(math.rad(90), 0, math.rad(90)))
							gravPool.Attachment.StrengthPool.Squash = NumberSequence.new(1)
						else
							gravLines.CFrame = sideFrame:ToWorldSpace(CFrame.new(0, 0, 0) * CFrame.Angles(math.rad(90), 0, math.rad(-90)))
							gravPool.CFrame = sideFrame:ToWorldSpace(CFrame.new(0.5, 5, 0) * CFrame.Angles(math.rad(90), 0, math.rad(-90)))
							gravPool.Attachment.StrengthPool.Squash = NumberSequence.new(1)
						end
						gravLines.Size = Vector3.new(20, 10, 30 * 1.3) * scale

local function getRotationBetween(u, v, axis)
	local axis = axis or Vector3.new(0,1,0) --backup axis
	local dot, uxv = u:Dot(v), u:Cross(v)
	if (dot < -0.99999) then return CFrame.fromAxisAngle(axis, math.pi) end
	return CFrame.new(0, 0, 0, uxv.x, uxv.y, uxv.z, 1 + dot)
end
local function toDegrees(...)
	local degs = {...}
	for i ,rad in pairs(degs) do
		degs[i] = math.deg(rad)
	end
	return table.unpack(degs)
end

	ccc.Playing = false
	ccc.Parent = slash.Primary

	-- Checking

		if spellname == "wahaj" then
			fireTrack:Play()
			fireTrack:AdjustSpeed(0.8)
			heat:FireServer("Wahaj", "WandRotation")
			local connection
			connection = fireTrack:GetMarkerReachedSignal("Straight"):Connect(function()
				heat:FireServer("Wahaj", "Start")
				connection:Disconnect()
			end)
			local connection2
			connection2 = fireTrack.Ended:Connect(function()
				heat:FireServer("Wahaj", "RotationFix")
				connection2:Disconnect()
			end)
		end
	elseif weapon then
		if spellname == "wahaj" then
			if MagoiBar.Value > 0 then
				-- Cost
				local Cost = 20
				if MagoiBar.Value - Cost > 0 then
					if weapon == "CutlassType1" then
						WahajWeaponTrack:Play()
						WahajWeaponTrack:AdjustSpeed(0.5)
						heat:FireServer("Wahaj", "SetUp")
						local connection 
						connection = WahajWeaponTrack:GetMarkerReachedSignal("PointOne"):Connect(function()
							heat:FireServer("Wahaj", "FirstSlash")
							connection:Disconnect()
						end)
						local connection2 
						connection2 = WahajWeaponTrack:GetMarkerReachedSignal("PointTwo"):Connect(function()
							heat:FireServer("Wahaj", "SecondSlash")
							connection2:Disconnect()
						end)
						local connection3
						connection3 = WahajWeaponTrack:GetMarkerReachedSignal("PointThree"):Connect(function()
							heat:FireServer("Wahaj", "ThirdSlash")
							connection3:Disconnect()
						end)
						local connect4
						connect4 = WahajWeaponTrack.Ended:Connect(function()
							heat:FireServer("Wahaj", "End")
							connect4:Disconnect()
						end)
						-- Animation Ender Listener
					end
				end
			end
		end
	end

sound.OnClientEvent:Connect(function(spellname)
	-- Check
	if Wand and chara.IntFold.CombatReady.Value == 1 then
		if spellname == "lahn" then
			-- Preventing Multiple Activation
			local casting = chara.Effects:FindFirstChild("Casting")
			local check = false
			if casting then
				if string.find(casting.Value, "Lahn") then
					check = true
				end
			end
			if not check then 
				-- Magoi Check
				local cost = 20
				if MagoiBar.Value - cost > 0 then		
					-- Playing Activation Animation
					Activate1Track:Play()
					Activate1Track:AdjustSpeed(0.75)
					sound:FireServer("Lahn", "Activation", cost)
					-- Listening For Follow-Up
					local con1
					con1 = chara:FindFirstChild("Effects").ChildAdded:Connect(function(child)
						if child.Name == "Casting" then 
							if child.Value == "placeholder" then
								con1:Disconnect()
								sound:FireServer("Lahn", "Composite", "Lahn-Placeholder")
							end
						end
					end)
					-- Ending
					local con2
					con2 = Activate1Track.Stopped:Connect(function()
						sound:FireServer("Lahn", "Ani1End")
						-- ani here
						con1:Disconnect()
						con2:Disconnect()
						local click = false
						-- Casting Mouse
						local rukhPart = Instance.new("Part")
						mouse.TargetFilter = rukhPart
						
						rukhPart.Name = "RukhPart"
						rukhPart.Size = Vector3.new(1,1,1)
						rukhPart.CanCollide = false
						rukhPart.Anchored = true
						rukhPart.Transparency = 1
						
						local particle2 = soundEffects.SoundRukh:Clone()
						particle2.Parent = rukhPart
						
						local billboard = game.ReplicatedFirst.GUIs.Magic.SoundCircleTrack:Clone()
						billboard.Adornee = rukhPart
						billboard.Parent = rukhPart
						
						rukhPart.Parent = workspace.Projectiles
						local con3
						local con4
						
						local con5
						con3 = rs.RenderStepped:Connect(function()
							local mag = (chara.HumanoidRootPart.Position - mouse.Hit.Position).Magnitude
							if mag < 200 then
								rukhPart.Position = mouse.Hit.Position
							end
						end)
						con4 = userinput.InputBegan:Connect(function(input, gpe)
							if gpe then return end
							if input.UserInputType == Enum.UserInputType.MouseButton1 then
								-- end ani.
								-- ani here
								-- Firing to Server
								sound:FireServer("Lahn", "Attack", rukhPart.Position)
								click = true
								-- Destroying RukhPart and Disconnecting
								rukhPart:Destroy()
								con2:Disconnect()
								con3:Disconnect()
								con4:Disconnect()
								con5:Disconnect()
							end
						end)
						-- Disconnecting if they have not clicked.
						delay(5, function()
							if not click then
								-- end ani.
								-- Firing to Server
								sound:FireServer("Lahn", "Misfire")
								-- Destroying RukhPart and Disconnecting
								rukhPart:Destroy()
								con2:Disconnect()
								con3:Disconnect()
								con4:Disconnect()
								con5:Disconnect()
							end
						end)
						
						-- Disconnecting if hit.
						con5 = chara.Effects.ChildAdded:Connect(function(child)
							if child.Value == "MagicHit" or child.Value == "LightHit" then
								-- Firing to Server
								sound:FireServer("Lahn", "Misfire")
								-- Destroying RukhPart and Disconnecting
								rukhPart:Destroy()
								con2:Disconnect()
								con3:Disconnect()
								con4:Disconnect()
								con5:Disconnect()
								
								-- Click for Delay
								click = false
							end
						end)
					end)
				else
					-- Spent Animaton
				end
			end
		end
	end
end)
